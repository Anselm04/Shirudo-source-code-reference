% A bar from #2 to #3 whose right end is rounded, at the top #4.
\newcommand\barend[4]{%
  \fill[#1] (#2,#4) -- ({#3-4},#4) arc[start angle=-90, end angle=0, radius=4]
    -- (#3,#4+12) arc[start angle=0, end angle=90, radius=4]
    -- (#2,#4+16) -- cycle;}
